### New Component: `frontend/src/components/RiskSummaryCard.tsx`

```typescript
import React from 'react';

interface Props {
  riskLevel: string;
  confidence: number;
  confidenceExplanation: string;
  riskExplanation: string;
}

const getRiskColor = (level: string) => {
  switch (level.toLowerCase()) {
    case 'low':
      return 'bg-green-100 border-green-300 text-green-900';
    case 'moderate':
      return 'bg-yellow-100 border-yellow-300 text-yellow-900';
    case 'high':
      return 'bg-red-100 border-red-300 text-red-900';
    default:
      return 'bg-gray-100 border-gray-300 text-gray-900';
  }
};

const getRiskIcon = (level: string) => {
  switch (level.toLowerCase()) {
    case 'low':
      return '🟢';
    case 'moderate':
      return '🟡';
    case 'high':
      return '🔴';
    default:
      return '⚪';
  }
};

export const RiskSummaryCard: React.FC<Props> = ({
  riskLevel,
  confidence,
  confidenceExplanation,
  riskExplanation,
}) => {
  const colorClass = getRiskColor(riskLevel);
  const icon = getRiskIcon(riskLevel);

  return (
    <div className={`border rounded-lg p-6 mb-6 ${colorClass}`}>
      <div className="mb-4">
        <div className="flex items-center gap-3 mb-2">
          <span className="text-2xl">{icon}</span>
          <h2 className="text-2xl font-bold capitalize">{riskLevel} Risk</h2>
        </div>
        <p className="text-sm opacity-75">Confidence: {confidence.toFixed(0)}%</p>
      </div>
      
      <div className="space-y-3">
        <div>
          <h3 className="font-semibold text-sm opacity-75 mb-1">Why this confidence level:</h3>
          <p className="leading-relaxed whitespace-pre-wrap">{confidenceExplanation}</p>
        </div>
        
        <div className="border-t opacity-50 pt-3">
          <h3 className="font-semibold text-sm opacity-75 mb-1">Why this risk level:</h3>
          <p className="leading-relaxed whitespace-pre-wrap">{riskExplanation}</p>
        </div>
      </div>
    </div>
  );
};
```

### New Component: `frontend/src/components/SpecialistCard.tsx`

```typescript
import React from 'react';

interface Props {
  specialist: string;
  explanation: string;
  urgency: string;
}

const getUrgencyBadge = (urgency: string) => {
  switch (urgency.toLowerCase()) {
    case 'immediate':
      return { text: 'IMMEDIATE', color: 'bg-red-600 text-white' };
    case '24-48 hours':
      return { text: 'Urgent (24-48 hrs)', color: 'bg-orange-500 text-white' };
    case '1-2 weeks':
      return { text: 'Within 1-2 weeks', color: 'bg-blue-500 text-white' };
    case 'routine':
      return { text: 'Routine', color: 'bg-gray-500 text-white' };
    default:
      return { text: urgency, color: 'bg-gray-500 text-white' };
  }
};

export const SpecialistCard: React.FC<Props> = ({ specialist, explanation, urgency }) => {
  const badge = getUrgencyBadge(urgency);

  return (
    <div className="border border-blue-200 rounded-lg p-6 mb-6 bg-blue-50">
      <div className="flex items-start justify-between mb-4">
        <div>
          <h2 className="text-xl font-bold text-blue-900 mb-1">👨‍⚕️ Recommended Specialist</h2>
          <p className="text-lg font-semibold text-blue-800">{specialist}</p>
        </div>
        <span className={`px-3 py-1 rounded-full text-sm font-semibold ${badge.color}`}>
          {badge.text}
        </span>
      </div>
      
      <div className="bg-white bg-opacity-50 rounded p-4">
        <p className="text-blue-900 leading-relaxed whitespace-pre-wrap">{explanation}</p>
      </div>
    </div>
  );
};
```
